  <script>
    (function() {
      // --- Elements ---
      const photoInput = document.getElementById('photoInput');

      const shareWaBtn = document.getElementById('shareWaBtn');

      let uploadedPhotoDataUrl = null;

      // --- Photo upload display ---
      photoInput.addEventListener('change', function(e) {
        const file = e.target.files[0];
        if (file) {
          fileNameDisplay.textContent = file.name.length > 20 ? file.name.substring(0, 18) + '…' : file.name;
          const reader = new FileReader();
          reader.onload = function(ev) {
            uploadedPhotoDataUrl = ev.target.result;
          };
          reader.readAsDataURL(file);
        } else {
          fileNameDisplay.textContent = 'No file selected';
          uploadedPhotoDataUrl = null;
        }
      });

      // --- Clear form ---
      clearFormBtn.addEventListener('click', function() {
        const inputs = document.querySelectorAll('#cvForm input, #cvForm textarea');
        inputs.forEach(input => {
          if (input.type !== 'file') input.value = '';
        });
        photoInput.value = '';
        fileNameDisplay.textContent = 'No file selected';
        uploadedPhotoDataUrl = null;
        document.getElementById('dob').value = '';
      });

      // --- Helper: get all form values ---
      function getFormData() {
        return {
          fullName: document.getElementById('fullName').value.trim(),
          contact: document.getElementById('contactNumber').value.trim(),
          email: document.getElementById('email').value.trim(),
          address: document.getElementById('address').value.trim(),
          jobPosition: document.getElementById('jobPosition').value.trim(),
          dob: document.getElementById('dob').value,
          education: document.getElementById('education').value.trim(),
          professional: document.getElementById('professional').value.trim(),
          workExperience: document.getElementById('workExperience').value.trim(),
          training: document.getElementById('training').value.trim(),
          skills: document.getElementById('skills').value.trim(),
          languages: document.getElementById('languages').value.trim(),
          certificates: document.getElementById('certificates').value.trim(),
        };
      }

      // --- Build PDF content (HTML) with professional single-page layout ---
      function buildPdfHtml() {
        const data = getFormData();
        const photoHtml = uploadedPhotoDataUrl 
          ? `<img src="${uploadedPhotoDataUrl}" style="width: 90px; height: 90px; border-radius: 50%; object-fit: cover; border: 2px solid #2a5588; margin-bottom: 10px;" alt="Photo">` 
          : '';

        let dobFormatted = data.dob;
        if (data.dob) {
          const d = new Date(data.dob);
          if (!isNaN(d)) {
            dobFormatted = d.toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' });
          }
        }

        const row = (labelEn, labelSi, value) => {
          if (!value || value.trim() === '') return '';
          return `<tr>
            <td style="padding: 5px 8px 5px 0; font-weight: 600; width: 38%; font-size: 10.5pt; color: #1e2e44; vertical-align: top;">${labelEn} <span style="font-weight: 400; color: #3a5a7a; font-size: 9pt;">(${labelSi})</span></td>
            <td style="padding: 5px 0; font-size: 10.5pt; color: #1a2638; vertical-align: top; white-space: pre-wrap; word-break: break-word;">${value.replace(/\n/g, '<br>')}</td>
          </tr>`;
        };

        return `
          <div style="font-family: 'Inter', Arial, sans-serif; max-width: 100%; background: white; border-radius: 12px; padding: 0;">
            <div style="display: flex; align-items: center; gap: 24px; margin-bottom: 18px; border-bottom: 3px solid #2a5588; padding-bottom: 14px;">
              <div style="flex-shrink: 0;">${photoHtml}</div>
              <div>
                <h1 style="font-size: 26pt; font-weight: 700; margin: 0 0 4px; color: #0f2c47; letter-spacing: -0.5px;">${data.fullName || 'Name Not Provided'}</h1>
                <p style="font-size: 12pt; font-weight: 500; margin: 0; color: #2a5588;">${data.jobPosition || 'Desired Position'}</p>
              </div>
            </div>

            <div style="display: flex; flex-wrap: wrap; gap: 12px 30px; margin-bottom: 20px; background: #f0f6fe; padding: 10px 16px; border-radius: 14px;">
              ${data.contact ? `<span style="font-size: 10.5pt;"><i style="color: #2a5588; margin-right: 6px;">📞</i> ${data.contact}</span>` : ''}
              ${data.email ? `<span style="font-size: 10.5pt;"><i style="color: #2a5588; margin-right: 6px;">✉️</i> ${data.email}</span>` : ''}
              ${dobFormatted ? `<span style="font-size: 10.5pt;"><i style="color: #2a5588; margin-right: 6px;">🎂</i> ${dobFormatted}</span>` : ''}
              ${data.address ? `<span style="font-size: 10.5pt;"><i style="color: #2a5588; margin-right: 6px;">🏠</i> ${data.address}</span>` : ''}
            </div>

            <table style="width: 100%; border-collapse: collapse; margin-top: 6px;">
              ${row('Educational Qualifications', 'අධ්‍යාපන සුදුසුකම්', data.education)}
              ${row('Professional Qualifications', 'වෘත්තීය සුදුසුකම්', data.professional)}
              ${row('Work Experience', 'රැකියා පළපුරුද්ද', data.workExperience)}
              ${row('Training & Internships', 'පුහුණුවීම් සහ Internship', data.training)}
              ${row('Skills', 'Skills', data.skills)}
              ${row('Language Proficiency', 'භාෂා දැනුම', data.languages)}
              ${row('Certificates & Awards', 'සහතික හා සම්මාන', data.certificates)}

              Prepared via HM CV MAKING SERVICES · www.hmcvmakingservices.lk · Contact: 072 814 4416
            </div>
          </div>
        `;
      }

      // --- Generate PDF (returns promise) ---
      function generatePdf() {
        return new Promise((resolve, reject) => {
          const htmlContent = buildPdfHtml();
          pdfPreview.innerHTML = htmlContent;

          const opt = {
            margin: [0.5, 0.5, 0.5, 0.5],
            filename: `CV_${getFormData().fullName || 'Details'}.pdf`,
            image: { type: 'jpeg', quality: 0.98 },
            html2canvas: { scale: 2, useCORS: true, letterRendering: true, logging: false },
            jsPDF: { unit: 'in', format: 'a4', orientation: 'portrait' }
          };

          html2pdf().set(opt).from(pdfPreview).save()
            .then(() => resolve())
            .catch((err) => {
              console.error('PDF error:', err);
              reject(err);
            });
        });
      }
